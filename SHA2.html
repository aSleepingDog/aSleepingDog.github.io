SHA2算法

上次更新:2026-10-13

概述

SHA2是美国NIST推出的第二代密码学散列函数，用于替代不安全的MD5和SHA1，目前被广泛用于需要消息校验和数字签名，几年前爆火虚拟货币——比特币，就是用的此算法（准确来说是SHA2-256版本）
作为分组散列函数，其将信息填充自固定的分组，然后设置一个初值和每个分块进行操作得到新的值，然后和后续的分块操作。

配置

SHA2的经典配置为256，224，384，512。他们的相关配置如下
配置 块大小（字节/比特） 输入限制（比特） 长度占位（字节/比特） 输出（字节/比特）
256 64/512 2^64-1 8/64 32/256
224 64/512 2^64-1 8/64 28/224
512 128/1024 2^128-1 16/128 64/512
384 128/1024 2^128-1 16/128 48/384

过程

接下来出现的数据长度单位均为比特

分组

消息M，其长度为size，分组长度为block_size，长度占位length_size，将M按照block_size一块块分组

规范化

对于最后一个分组（长度last_size），在这个分组后加上一个比特1（此时长度last_size+1）。
判断block_size-(last_size+1)和length_size的大小
如果相等，则直接按大端方式把size写到此分组末尾。
如果小于，则不断填充比特0，直到相等，然后按照相等方式填充。
如果大于，则此分组剩下的位全部填充比特0。然后新开一个分组（长度remain_size），继续填充比特0， 直到block_size-remain_size等于length_size然后按照相等方式填充。

uint128_t operator<<(const uint128_t& a, uint64_t shift)
{
    shift %= 128;
    if(shift >= 64)
    {
        return {0, a[0] << (shift - 64)};
    }
    return {a[0]<<shift, (a[1]<<shift) | (a[0]>>(64-shift))};
}
void u64add(uint64_t carry_in, uint64_t a, uint64_t b, uint64_t* carry, uint64_t* result)
{
    uint64_t low = (a & 0xFFFFFFFF) + (b & 0xFFFFFFFF) + carry_in;
    uint64_t high = (a >> 32) + (b >> 32) + (low >> 32);
    *carry = high >> 32;
    *result = high << 32 | (low & 0xFFFFFFFF);
}
uint128_t operator+(const uint128_t& a, const uint128_t& b)
{
    uint128_t result;
    uint64_t carry = 0;
    u64add(carry, a[0], b[0], &carry, &result[0]);
    u64add(carry, a[1], b[1], &carry, &result[1]);
    return result;
}
void u64sub(uint64_t borrow_in, uint64_t a, uint64_t b, uint64_t* borrow, uint64_t* result)
{
    uint64_t carry = b == 0xFFFFFFFFFFFFFFFF && borrow_in == 1 ? 1 : 0;
    b += borrow_in;
    bool com = a >= b;
    *borrow = carry | !com;
    *result = com ? a - b : (~(b - a) + 1);
}
uint128_t operator-(const uint128_t& a, const uint128_t& b)
{
    uint128_t result;
    uint64_t borrow = 0;
    u64sub(0, a[0], b[0], &borrow, &result[0]);
    u64sub(borrow, a[1], b[1], &borrow, &result[1]);
    return result;
}
uint64_t square_root(uint64_t p_)
{
    uint128_t p = {p_,0};
    uint128_t r = {p_/2,0};
    while(r[0]*r[0] > p[0])
    {
        r[0]--;
    }
    p[0] -= r[0]*r[0];
    for(uint64_t i = 0; i < 64; i++)
    {
        p = p << 2;
        uint128_t t = (r<<2);
        t[0]|=1;
        r = r << 1;
        if(p >= t)
        {
            p = p - t;
            r[0]|=1;
        }
    }
    return r[0];
}

def sqrt(a: int):
    r0 = a//2
    r1 = 0;
    while r0!=r1:
        r1 = r0
        r0 = (r0 + a//r0)//2
    return r0

分组扩展

接下来以>>和<<表示逻辑移位（超出部分丢弃，不足部分补0），以>>>和<<<表示循环移位（超出补分补到另一边）
字和字的加法如果有进位溢出则直接切断丢弃，用|表示大端拼接（比如FE = F | E）
将单个分组重新解释为若干个32位字（224和256）或64位字（384和512），以当前分组末尾位置（即偏移量16的字），以大端方式取出偏移量为1，14，0，9的字
命名为$s_0$,$s_1$,$s_2$,$s_3$,然后做以下操作
$$s_0 = s_0 \ggg 7 \oplus s_0 \ggg 18 \oplus s_0 \gg 3$$
$$s_1 = s_1 \ggg 17 \oplus s_1 \ggg 19 \oplus s_1 \gg 10$$
$$a = s_0 + s_1 + s_2 + s_3$$
然后按照大端方式将新的字a，填充到分组末尾
之后依照上述方式补充分组至64个字（224和256）和80个字（384和512）

std::strong_ordering operator<=>(const uint192_t& a, const uint192_t& b)
{
    if (a[2] != b[2])

uint192_t operator<<(const uint192_t& a, uint64_t shift)
{
    shift %= 192;
    uint192_t result;
    if (shift >= 128)
    {
        shift -= 128;
        result[2] = a[0] << shift;
    }
    else if (shift >= 64)
    {
        shift -= 64;
        result[1] = a[0] << shift;
        result[2] = a[1] << shift | (a[0] >> (64 - shift));
    }
    else if (shift > 0)

        result[1] = a[1] << shift | (a[0] >> (64 - shift));
        result[2] = a[2] << shift | (a[1] >> (64 - shift));
    }
    else
    {
        result = a;
    }
    return result;
}

void u64mul(uint64_t a, uint64_t b, uint64_t* result_high, uint64_t* result_low)
{
    uint32_t a0 = a & 0xffffffff;
    uint32_t a1 = a >> 32;
    uint32_t b0 = b & 0xffffffff;
    uint32_t b1 = b >> 32;

    uint64_t low = (uint64_t)a0 * b0;
    uint64_t high = (uint64_t)a1 * b1;
    uint64_t mid0 = (uint64_t)a0 * b1;
    uint64_t mid1 = (uint64_t)a1 * b0;
    if (mid0 > 0xffffffffffffffff - mid1)
    {
        high += 0x100000000;
    }
    mid0 += mid1;
    high += (mid0) >> 32;
    if (low > 0xffffffffffffffff - (mid0 << 32))
    {
        high++;
    }
    low += mid0 << 32;

    *result_high = high;
    *result_low = low;
}
uint192_t operator*(const uint192_t& a, uint64_t b)
{
    uint64_t now_h = 0, now_l = 0, carry = 0, i = 0;
    uint192_t result;
    for (; i < a.size(); ++i)
    {
        u64mul(a[i], b, &now_h, &now_l);
        u64add(0, carry, now_l, &carry, &now_l);
        carry += now_h;
        result[i] = now_l;
    }
    return result;
}
uint192_t operator*(const uint192_t& a, const uint192_t& b)
{
    uint192_t r0 = a * b[0];
    uint192_t r1 = a * b[1];
    uint192_t r2 = a * b[2];
    r1[2] = r1[1];
    r1[1] = r1[0];
    r1[0] = 0;
    r2[2] = r2[0];
    r2[1] = 0;
    r2[0] = 0;
    return r0 + r1 + r2;
}
uint64_t cube_root(uint64_t p_)
{
    uint192_t p = { p_, 0, 0 };
    uint192_t r = { 10, 0, 0 };
    while (r[0] * r[0] * r[0] > p[0])
    {
        r[0]--;
    }
    p[0] -= r[0] * r[0] * r[0];
    for (int i = 0; i < 64; i++)
    {
        p = p << 3;
        uint192_t t = r << 2;
        t[0] |= 2;
        t = ((r << 1) + r) * t;
        t[0] |= 1;
        r = r << 1;
        if (t <= p)
        {
            p = p - t;
            r[0] |= 1;
        }
    }
    return r[0];
}

def cuber(a: int):
    r = 10
    while r**3 > a:
        r -= 1
    a -= r**3
    for _ in range(64):
        a <<= 3
        t = ((r<<1)+r)*((r<<2)+2)+1
        r <<= 1
        if a >= t:
            a -= t
            r += 1
    return r

混淆计算

接下来计算64轮（224和256）或80轮（224和256），记当前轮为$i$,其取值从0开始到63（224和256）或79（224和256）
取扩展后分组的偏移量为$i$的$p$位（$p=32$ 224和256）（$p=64$ 384和512），记为$w$。再取常量$k=K_{i}$
将初始值$I$分为$I^{p}_0|I^{p}_1|I^{p}_2|I^{p}_3|I^{p}_4|I^{p}_5|I^{p}_6|I^{p}_7|I^{p}_8$，分别记为$a|b|c|d|e|f|g|h$
然后依次做以下操作
$$ S_0 = a \ggg 2 \oplus a \ggg 13 \oplus a \ggg 22 $$
$$ M = (a \& b) \oplus (a \& c) \oplus (b \& c) $$
$$ T_2 = S_0 + M $$
$$ S_1 = e \ggg 6 \oplus e \ggg 11 \oplus e \ggg 25 $$
$$ N = (e \& f) \oplus ((\sim e) \& g) $$
$$ T_1 = h + S_1 + N + k + w $$
通过上述变量更新$d$和$h$的值
$$ h = T_1 + T_2 $$
$$ d = T_1 + d $$
再轮换更新值
$$ a|b|c|d|e|f|g|h \Rightarrow h|a|b|c|d|e|f|g = I' $$
将$I'$作为下一轮的初始值，继续计算直到达到轮次
将最后一轮的$I'$和初始值$I$，做下列操作得到$H$
$$ H = I^{p}_0 + I'^{p}_0|I^{p}_1 + I'^{p}_1|I^{p}_2 + I'^{p}_2|I^{p}_3 + I'^{p}_3|I^{p}_4 + I'^{p}_4|I^{p}_5 + I'^{p}_5|I^{p}_6 + I'^{p}_6|I^{p}_7 + I'^{p}_7 $$
$H$则为消息的散列值

uint8_t index(const std::vector<uint8_t>& front, const std::vector<uint8_t>& back, uint64_t i)
{
    if (i < front.size())
        return front[i];
    else
        return back[i - front.size()];
}
uint32_t rotr(uint32_t x, uint32_t n)
{
    n %= 32;
    return (x >> n) | (x << (32 - n));
}
std::vector<uint8_t> sha2_256(const std::vector<uint8_t>& data)
{
    std::vector<uint8_t> last = { 0x80 };
    while ((last.size() + data.size()) % 64 != 56)
    {
        last.push_back(0);
    }
    uint64_t len = data.size() * 8;
    for (int i = 0; i < 8; i++)
    {
        last.push_back((len >> (56 - i * 8)) & 0xff);
    }
    std::array<uint32_t, 64> words;
    std::array<uint32_t, 8> init = { 0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19 };
    std::array<uint32_t, 8> hash = init;
    uint32_t& a = hash[0];
    uint32_t& b = hash[1];
    uint32_t& c = hash[2];
    uint32_t& d = hash[3];
    uint32_t& e = hash[4];
    uint32_t& f = hash[5];
    uint32_t& g = hash[6];
    uint32_t& h = hash[7];
    for (uint64_t i = 0; i < data.size() + last.size(); i += 64)
    {
        uint64_t j = 0;
        for (j = 0; j < 64; ++j)
        {
            if (j % 4 == 0) words[j / 4] = 0;
            words[j / 4] <<= 8;
            words[j / 4] |= index(data, last, i + j);
        }
        for (j = 0; j < 48; ++j)
        {
            uint32_t s0 = words[j + 1];
            uint32_t s1 = words[j + 14];
            uint32_t s2 = words[j + 0];
            uint32_t s3 = words[j + 9];
            s0 = rotr(s0, 7) ^ rotr(s0, 18) ^ (s0 >> 3);
            s1 = rotr(s1, 17) ^ rotr(s1, 19) ^ (s1 >> 10);
            words[j + 16] = s0 + s1 + s2 + s3;
        }
        for (j = 0; j < 64; ++j)
        {
            uint32_t w = words[j];
            uint32_t k = ks[j];
            uint32_t s0 = rotr(a, 2) ^ rotr(a, 13) ^ rotr(a, 22);
            uint32_t m = (a & b) ^ (a & c) ^ (b & c);
            uint32_t t2 = s0 + m;
            uint32_t s1 = rotr(e, 6) ^ rotr(e, 11) ^ rotr(e, 25);
            uint32_t ch = (e & f) ^ (~e & g);
            uint32_t t1 = h + s1 + ch + k + w;
            
            h = t1 + t2;
            d += t1;
            uint32_t x = h;
            h = g;
            g = f;
            f = e;
            e = d;
            d = c;
            c = b;
            b = a;
            a = x;
        }
        for (j = 0; j < 8; ++j)
        {
            hash[j] += init[j];
            init[j] = hash[j];
        }
    }
    std::vector<uint8_t> result;
    for (uint64_t i = 0; i < 8; i++)
    {
        result.push_back(hash[i] >> 24);
        result.push_back(hash[i] >> 16);
        result.push_back(hash[i] >> 8);
        result.push_back(hash[i]);
    }
    return result;
}
